## Create the passkey

```ts
import { createPasskeyWithPrfOutput } from "@category-labs/mera";

const rpId = location.hostname;

const created = await createPasskeyWithPrfOutput({
  rp: { id: rpId, name: "Example" },
  user: { name: "account@example.com", displayName: "Example account" },
});
```

Every call creates a new passkey.

## Remember the credential

The app can store the credential metadata and pass it back on the next sign-in, so the browser reuses the same passkey instead of offering a choice.

```ts
import type { createPasskeyWithPrfOutput } from "@category-labs/mera";

declare const created: createPasskeyWithPrfOutput.Result;

// ---cut---
localStorage.setItem(
  "app.derivedCredential",
  JSON.stringify({
    credentialId: created.credentialId,
    transports: created.transports,
  }),
);
```

`transports` is optional and only a hint: the browser uses it to reach the authenticator directly (a platform prompt for a platform passkey, a QR flow for a phone) instead of offering every option.

## Sign in

```ts
const rpId = "account.example.com";

// ---cut---
import { getPasskeyPrfOutput } from "@category-labs/mera";

const stored = localStorage.getItem("app.derivedCredential");
const known = stored ? JSON.parse(stored) : undefined;

const { prfOutput, credentialId } = await getPasskeyPrfOutput({
  rpId,
  credential: known,
});

// With no stored record, the browser may have used any discoverable passkey.
const record =
  known?.credentialId === credentialId ? known : { credentialId };
localStorage.setItem("app.derivedCredential", JSON.stringify(record));
```

## Derive numbered accounts

`deriveEvmPrivateKey` and `deriveSolanaPrivateKey` number accounts by `accountIndex`. These helpers open one account of each kind:

```ts
import {
  createEd25519SigningSession,
  createSecp256k1SigningSession,
  deriveEvmPrivateKey,
  deriveSolanaPrivateKey,
  getEvmAddress,
  getSolanaAddress,
} from "@category-labs/mera";

function openEvmAccount(prfOutput: Uint8Array, accountIndex: number) {
  const privateKey = deriveEvmPrivateKey({ prfOutput, accountIndex });
  const session = createSecp256k1SigningSession({ privateKey });
  privateKey.fill(0);
  return { session, address: getEvmAddress(session.publicKey) };
}

function openSolanaAccount(prfOutput: Uint8Array, accountIndex: number) {
  const privateKey = deriveSolanaPrivateKey({ prfOutput, accountIndex });
  const session = createEd25519SigningSession({ privateKey });
  privateKey.fill(0);
  return { session, address: getSolanaAddress(session.publicKey) };
}
```

Each session copies the key, so the helpers zero the derived key right away.

Build the account list from these helpers. The PRF output reproduces every account, so it is zeroed once they are open:

```ts
import type {
  Ed25519SigningSession,
  EvmAddress,
  Secp256k1SigningSession,
} from "@category-labs/mera";

declare const prfOutput: Uint8Array;
declare function openEvmAccount(
  prfOutput: Uint8Array,
  accountIndex: number,
): { session: Secp256k1SigningSession; address: EvmAddress };
declare function openSolanaAccount(
  prfOutput: Uint8Array,
  accountIndex: number,
): { session: Ed25519SigningSession; address: string };

// ---cut---
const accounts = [
  openEvmAccount(prfOutput, 0),
  openSolanaAccount(prfOutput, 0),
];
prfOutput.fill(0);
```

## End the sessions

```ts
import type {
  Ed25519SigningSession,
  Secp256k1SigningSession,
} from "@category-labs/mera";

declare const accounts: Array<{
  session: Ed25519SigningSession | Secp256k1SigningSession;
}>;

// ---cut---
for (const account of accounts) {
  account.session.end();
}
```

Ending a session zeroes the private keys it owns.
